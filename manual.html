ก่อนเปิดรับสมัครปีใหม่ ต้องลบข้อมูลเก่าในฐานข้อมูล (ห้ามลบตารางสีและตารางคณะเด็ดขาด)

วิธีทำ: ให้เปิด phpMyAdmin เลือกฐานข้อมูล hukangame แล้วกดไปที่แท็บ SQL จากนั้นวางคำสั่งต่อไปนี้แล้วกด Go (หรือรันทีละบรรทัด):

-- 1. ลบข้อมูลการลงทะเบียนแข่งทั้งหมด
DELETE FROM `sport_registration`; -- ใช้ DELETE เพราะมี foreign key จาก sport_attendance (ลบประวัติเช็คชื่อให้อัตโนมัติ)

-- 2. ลบข้อมูลรูปภาพหน้าตรง/บัตรนักศึกษา
TRUNCATE TABLE `image`;

-- 3. ลบข้อมูลไซซ์เสื้อ
TRUNCATE TABLE `player_size`;

-- 4. ลบข้อมูลนักศึกษา (เฉพาะปีนั้นๆ เพื่อให้ระบบเคลียร์)
TRUNCATE TABLE `student`;

-- 5. (ถ้าต้องการลบประวัติการจัดตารางแข่งเก่า)
DELETE FROM `match_schedule`;
หมายเหตุ: หากตาราง faculty (รายชื่อคณะ) หรือตาราง color_team (สี) มีการเปลี่ยนข้อมูล ให้ดูหัวข้อ 5.3 ด้านล่าง

🟢 การตั้งค่า Google Sheets & Apps Script (กรณีย้ายอีเมล)

หากเปลี่ยนอีเมลแอดมินใหม่ ต้องสร้างสคริปต์รับข้อมูลใหม่ เพื่อให้ส่งข้อมูลจากหน้าเว็บเข้าชีตได้

7.1 ขั้นตอนการสร้าง Spreadsheet และ Script

  • ใช้ Gmail ใหม่ ล็อกอินเข้า Google Sheets และกดสร้างชีตใหม่ (Blank)แนะนำให้เป็น (@gmail.com)
  • ตั้งชื่อชีตให้เรียบร้อย (เช่น HukanGame_2027)
  • ที่แถบเมนูด้านบน กด ส่วนขยาย (Extensions) > Apps Script
  • ลบโค้ดเดิมที่มีอยู่ทิ้งให้หมด แล้วก๊อปปี้โค้ดด้านล่างนี้ไปวาง:
function doPost(e) {
  var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
  try {
    var rawData = e.postData.contents;
    var payload = JSON.parse(rawData);
    var dataRows = payload.data;

    // สร้างหัวตารางถ้ายกเลิกชีตว่างเปล่า
    if (sheet.getLastRow() === 0) {
      sheet.appendRow(["รหัสนักศึกษา", "ชื่อ-สกุล", "เบอร์โทร", "คณะ", "สี", "กีฬา", "ประเภท", "ไซซ์เสื้อ"]);
    }

    var rowsToAppend = [];
    for (var i = 0; i < dataRows.length; i++) {
      var r = dataRows[i];
      rowsToAppend.push([r.id, r.name, r.tel, r.faculty, r.color, r.sport, r.category, r.size]);
    }

    // นำข้อมูลลงชีตแบบรวดเดียว (Batch)
    sheet.getRange(sheet.getLastRow() + 1, 1, rowsToAppend.length, rowsToAppend[0].length).setValues(rowsToAppend);

    return ContentService.createTextOutput(JSON.stringify({"status": "success"}))
                         .setMimeType(ContentService.MimeType.JSON);
  } catch (err) {
    return ContentService.createTextOutput(JSON.stringify({"status": "error", "msg": err.toString()}))
                         .setMimeType(ContentService.MimeType.JSON);
  }
}

7.2 การ Deploy เพื่อเอา URL ไปใช้

  • กดปุ่ม บันทึก (Save) ที่เป็นรูปแผ่นดิสก์
  • กดปุ่มสีฟ้าด้านขวาบน การทำให้ใช้งานได้ (Deploy) > การสร้างรายการใหม่ (New deployment)
  • ตรง "เลือกประเภท" ให้คลิกรูปฟันเฟือง เลือก แอปพลิเคชันเว็บ (Web app)
  • ตรง "ผู้ที่มีสิทธิ์เข้าถึง (Who has access)" สำคัญมาก! ต้องเลือกเป็น: ทุกคน (Anyone) ไม่เช่นนั้นเว็บจะ Sync ไม่ได้
  • กด Deploy และให้สิทธิ์บัญชี (Authorize Access) จนเสร็จสิ้น
  • ก๊อปปี้ URL ของ Web app ที่ได้ นำไปวางในไฟล์ c:\xampp\htdocs\a\allstudent2.php (ดูวิธีการแก้ที่หน้า 5 ข้อ 5.1)
💡 คำแนะนำ: หากมีการแก้โค้ดใน Apps Script ในอนาคต การกด Deploy ครั้งต่อไปต้องเลือกเป็น Manage deployments และกดรูปดินสอแก้ไข เพื่อให้ URL ไม่เปลี่ยน (จะได้ไม่ต้องไปแก้ในไฟล์ PHP บ่อยๆ)
